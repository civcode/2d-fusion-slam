cmake_minimum_required(VERSION 3.25)
project(fusion_dependency_smoke LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 3.4 REQUIRED CONFIG)
find_package(Catch2 3.16 REQUIRED CONFIG)
find_package(Ceres 2.2 REQUIRED CONFIG)
find_package(GTSAM 4.3 REQUIRED CONFIG)
if(NOT TARGET gtsam)
  message(FATAL_ERROR "Installed GTSAM package did not export the gtsam target")
endif()
add_executable(fusion_dependency_smoke dependency_smoke.cpp)
target_link_libraries(fusion_dependency_smoke PRIVATE Eigen3::Eigen Catch2::Catch2WithMain Ceres::ceres gtsam)
add_test(NAME dependency_link_runtime COMMAND fusion_dependency_smoke)
